    async function addSnackWithAI() {
      const snackText = document.getElementById('snackInput').value.trim();
      const apiKey = document.getElementById('apiKeyInput').value.trim();
      if (!snackText) return;

      let estCalories = 150; // Fallback default estimate

      if (apiKey) {
        try {
          const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
              contents: [{ parts: [{ text: `Estimate calorie for this snack: "${snackText}". Return ONLY an integer digit number of calories.` }] }]
            })
          });
          const data = await res.json();
          const text = data.candidates[0].content.parts[0].text;
          const match = text.match(/\d+/);
          if (match) estCalories = parseInt(match[0]);
        } catch (e) {
          console.error("AI Snack estimation failed, using standard default.", e);
        }
      }

      appState.addedSnacks.push({ name: snackText, kcal: estCalories });
      document.getElementById('snackInput').value = '';
      renderSnacks();
    }

    function renderSnacks() {
      const list = document.getElementById('snackList');

    }

    // --- SECTION 3: PPG PULSE RATE MEASUREMENT ---
    let videoStream = null;
    let ppgInterval = null;

    async function togglePPG() {
      const btn = document.getElementById('startPpgBtn');
      const status = document.getElementById('ppgStatus');

      if (videoStream) {
        clearInterval(ppgInterval);
        videoStream.getTracks().forEach(track => track.stop());
        videoStream = null;
        btn.innerText = "Start Measurement";
        status.innerText = `Measurement Saved: ${appState.pulseRate} BPM`;
        return;
      }

      try {
        videoStream = await navigator.mediaDevices.getUserMedia({
          video: { facingMode: 'environment', torch: true }
        });
        const video = document.getElementById('video');
        video.srcObject = videoStream;
        btn.innerText = "Stop Measurement";
        status.innerText = "Measuring... Hold finger still.";

        // Real PPG frame red-channel intensity extraction logic
        const canvas = document.createElement('canvas');
        const ctx = canvas.getContext('2d');
        let frames = [];

        ppgInterval = setInterval(() => {
          if (video.videoWidth > 0) {
            canvas.width = video.videoWidth;
            canvas.height = video.videoHeight;
            ctx.drawImage(video, 0, 0);
            const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height);
            let redSum = 0;
            for (let i = 0; i < imgData.data.length; i += 16) {
              redSum += imgData.data[i];
            }
            frames.push(redSum);
            if (frames.length > 100) frames.shift();

            // Calculate peaks for PPG pulse
            if (frames.length >= 60) {
              let peaks = 0;
              for (let i = 1; i < frames.length - 1; i++) {
                if (frames[i] > frames[i-1] && frames[i] > frames[i+1]) peaks++;
              }
              const calculatedBpm = Math.min(Math.max(peaks * 12, 60), 120);
              appState.pulseRate = calculatedBpm;
              document.getElementById('ekgBpm').innerText = calculatedBpm;
            }
          }
        }, 100);

      } catch (err) {
        status.innerText = "Camera access error or light unavailable.";
      }
    }

    // --- SECTION 5: HEALTH CALCULATIONS & AI ---
    function calculateOverallHealth() {
      // 1. Dietary Calorie Balance Score (20%)
      const dayMenu = menuData[appState.dayOfMonth] || menuData[1];
      let eatenKcal = 0;
      for (let key in dayMenu) {
        if (appState.meals[key]) eatenKcal += dayMenu[key];
      }
      appState.addedSnacks.forEach(s => eatenKcal += s.kcal);
      
      const idealKcal = 2200;
      const dietDiff = Math.abs(eatenKcal - idealKcal);
      const dietScore = Math.max(0, 100 - (dietDiff / 15));

      // 2. BMI Score (20%)
      const bmi = calculateBMI();
      let bmiScore = 100 - (Math.abs(bmi - 21.5) * 8);
      bmiScore = Math.max(0, Math.min(100, bmiScore));

      // 3. Hydration Score (15%)
      const hydrationScore = Math.min(100, (appState.glassesDrunk / 8) * 100);

      // 4. Pulse Rate Score (10%)
      const pulse = appState.pulseRate;
      let pulseScore = 100 - (Math.abs(pulse - 70) * 2);
      pulseScore = Math.max(0, Math.min(100, pulseScore));

      // 5. Stair Activity Score (20%)
      let totalStairKcal = 0;
      let totalClimbs = 0;
      const allStairs = [...testedStairsData, ...genericStairsData];
      allStairs.forEach(s => {
        const count = appState.stairCounts[s.id] || 0;
        totalClimbs += count;
        totalStairKcal += count * s.energyKcal;
      });
      const stairScore = Math.min(100, totalClimbs * 10);

      // 6. Stress Level Score (15%)
      const avgStress = appState.stressScores.reduce((a,b) => a+b, 0) / 5;
      const stressScore = (avgStress / 5) * 100;

      // Final Weighted Score calculation
      const finalScore = Math.round(
        (dietScore * 0.20) +
        (bmiScore * 0.20) +
        (hydrationScore * 0.15) +
        (pulseScore * 0.10) +
        (stairScore * 0.20) +
        (stressScore * 0.15)
      );

      // Update UI elements
      document.getElementById('scoreCircle').style.setProperty('--score', finalScore);
      document.getElementById('scoreValue').innerText = finalScore;
      document.getElementById('ovDiet').innerText = `${eatenKcal} kcal`;
      document.getElementById('ovBMI').innerText = `${bmi}`;
      document.getElementById('ovHydration').innerText = `${appState.glassesDrunk * 250} / 2000 ml`;
      document.getElementById('ovBurned').innerText = `${totalStairKcal.toFixed(2)} kcal`;
      document.getElementById('ovStress').innerText = `${avgStress.toFixed(1)} / 5`;
    }

    // --- EKG CANVAS ANIMATION ---
    function initEKGCanvas() {
      const canvas = document.getElementById('ekgCanvas');
      const ctx = canvas.getContext('2d');
      let x = 0;

      function resize() {
        canvas.width = canvas.parentElement.clientWidth;
        canvas.height = 120;
      }
      resize();
      window.addEventListener('resize', resize);

      function drawEKG() {
        ctx.fillStyle = 'rgba(0, 0, 0, 0.05)';
        ctx.fillRect(0, 0, canvas.width, canvas.height);

        ctx.strokeStyle = '#00ff66';
        ctx.lineWidth = 2;
        ctx.beginPath();
        
        const midY = canvas.height / 2;
        let y = midY;

        // Simulated PQRST ECG waveform signal
        const cycle = x % 100;
        if (cycle > 40 && cycle < 45) y = midY - 10; // P Wave
        else if (cycle >= 45 && cycle < 48) y = midY + 10; // Q Wave
        else if (cycle >= 48 && cycle < 52) y = midY - 45; // R Wave
        else if (cycle >= 52 && cycle < 55) y = midY + 20; // S Wave
        else if (cycle >= 65 && cycle < 75) y = midY - 15; // T Wave

        ctx.moveTo(x, midY);
        ctx.lineTo(x + 1, y);
        ctx.stroke();

        x = (x + 2) % canvas.width;
        requestAnimationFrame(drawEKG);
      }
      drawEKG();
    }
